43def clean_with_isolation_forest( 

44 input_files: Sequence[tfs.TfsDataFrame], meas_input: DotDict, plane: str, 

45) -> Sequence[tfs.TfsDataFrame]: 

46 """ 

47 Runs isolation forest anomaly detection and removes flagged BPMs from input files. 

48 For every file where a cleaning is performed, a TfsDataFrame with information on 

49 the identified and cleaned BPMs is written to disk in an adjacent location in the 

50 corresponding output dir. 

51 

52 Args: 

53 input_files: list of measurement DataFrames. 

54 meas_input: `OpticsInput` object containing analysis settings. 

55 plane: marking the horizontal or vertical plane, **X** or **Y**. 

56 

57 Returns: 

58 The input files with anomalous BPMs removed. 

59 """ 

60 bad_bpms = identify_bad_bpms(meas_input, input_files, plane) 

61 input_files = remove_bad_bpms(input_files, list(set(bad_bpms.NAME)), plane) 

62 LOGGER.info(str(list(set(bad_bpms.NAME)))) 

63 if meas_input.outputdir is not None: 

64 tfs.write(Path(meas_input.outputdir)/ f"bad_bpms_iforest_{plane.lower()}.tfs", bad_bpms) 

65 return input_files

68def identify_bad_bpms( 

69 meas_input: DotDict, 

70 input_files: Sequence[tfs.TfsDataFrame], 

71 plane: str, 

72) -> pd.DataFrame: 

73 """ 

74 Identifies anomalous BPMs across arc and IR regions using isolation forest. 

75 

76 Args: 

77 meas_input: `OpticsInput` object containing analysis settings. 

78 input_files: list of measurement DataFrames. 

79 plane: marking the horizontal or vertical plane, **X** or **Y**. 

80 

81 Returns: 

82 A `DataFrame` listing the detected bad BPMs with their significant features and scores. 

83 """ 

84 bpm_data = pd.concat( 

85 [tfs_df[["NAME", f"TUNE{plane}", "NOISE_SCALED", f"AMP{plane}"]] 

86 for tfs_df in input_files] 

87 ) 

88 arc_bpm_data, ir_bpm_data = get_data_for_clustering(bpm_data, plane, meas_input.accelerator) 

89 return pd.concat([identify_single_cluster_bad_bpms(bpm_data, ARCS_CONT, arc_bpm_data, plane), 

90 identify_single_cluster_bad_bpms(bpm_data, IRS_CONT, ir_bpm_data, plane)])

93def identify_single_cluster_bad_bpms( 

94 bpm_tfs_data: pd.DataFrame, 

95 cont: float, 

96 data_for_clustering: pd.DataFrame, 

97 plane: str, 

98) -> pd.DataFrame: 

99 """Runs isolation forest on a single cluster and returns the significant features of detected anomalies. 

100 

101 Args: 

102 bpm_tfs_data: concatenated BPM data from all input files. 

103 cont: contamination parameter for the isolation forest. 

104 data_for_clustering: normalised feature data for clustering. 

105 plane: marking the horizontal or vertical plane, **X** or **Y**. 

106 

107 Returns: 

108 A `DataFrame` with anomalous BPM names, their most significant feature, and anomaly scores. 

109 """ 

110 bad_bpms, good_bpms, bad_bpms_scores = detect_anomalies(cont, data_for_clustering, plane) 

111 bpm_tfs_data, data_for_clustering, bad_bpms, good_bpms = [ 

112 reassign_index(data) for data in (bpm_tfs_data, data_for_clustering, bad_bpms, good_bpms) 

113 ] 

114 signif_feature = get_significant_features(bpm_tfs_data, data_for_clustering, bad_bpms, 

115 good_bpms, plane) 

116 signif_feature.loc[:, "SCORE"] = bad_bpms_scores 

117 return signif_feature

120def reassign_index(data: pd.DataFrame) -> pd.DataFrame: 

121 """Resets the index of the DataFrame to a sequential integer range.""" 

122 data["NEW_INDEX"] = range(len(data.NAME)) 

123 return data.set_index("NEW_INDEX")

126def get_significant_features( 

127 bpm_tfs_data: pd.DataFrame, 

128 data_for_clustering: pd.DataFrame, 

129 bad_bpms: pd.DataFrame, 

130 good_bpms: pd.DataFrame, 

131 plane: str, 

132) -> pd.DataFrame: 

133 """Determines the most significant feature for each anomalous BPM. 

134 

135 For each bad BPM, finds the feature (tune, noise, or amplitude) with the 

136 largest deviation from the mean of good BPMs. 

137 

138 Args: 

139 bpm_tfs_data: concatenated BPM data from all input files. 

140 data_for_clustering: normalised feature data used for clustering. 

141 bad_bpms: DataFrame of BPMs flagged as anomalous. 

142 good_bpms: DataFrame of BPMs considered normal. 

143 plane: marking the horizontal or vertical plane, **X** or **Y**. 

144 

145 Returns: 

146 A `DataFrame` indexed like ``bad_bpms`` with columns NAME, FEATURE, VALUE, and AVG. 

147 """ 

148 features_df = pd.DataFrame(index=bad_bpms.index) 

149 for index in bad_bpms.index: 

150 max_dist = max([(abs(data_for_clustering.loc[index, col] - 

151 good_bpms.loc[:, col].mean()), col) 

152 for col in [f"TUNE{plane}", "NOISE_SCALED", f"AMP{plane}"]]) 

153 max_dist, sig_col = max_dist 

154 features_df.loc[index, "NAME"] = bad_bpms.loc[index, "NAME"] 

155 features_df.loc[index, FEATURE] = sig_col 

156 features_df.loc[index, "VALUE"] = bpm_tfs_data.loc[index, sig_col] 

157 features_df.loc[index, "AVG"] = np.mean(bpm_tfs_data.loc[good_bpms.index][sig_col]) 

158 return features_df

161def detect_anomalies( 

162 contamination: float, data: pd.DataFrame, plane: str, 

163) -> tuple[pd.DataFrame, pd.DataFrame, np.ndarray]: 

164 """Fits an isolation forest and separates data into anomalous and normal BPMs. 

165 

166 Args: 

167 contamination: expected proportion of outliers in the data. 

168 data: normalised BPM feature data. 

169 plane: marking the horizontal or vertical plane, **X** or **Y**. 

170 

171 Returns: 

172 A tuple of (bad_bpms, good_bpms, bad_bpms_scores). 

173 """ 

174 iforest = IsolationForest(n_estimators=100, max_samples='auto', 

175 contamination=contamination, max_features=1.0, 

176 bootstrap=False) 

177 features = data[[f"TUNE{plane}", "NOISE_SCALED", f"AMP{plane}"]] 

178 iforest.fit(features) 

179 labels = iforest.predict(features) 

180 bad_bpms = data.iloc[np.where(labels == -1)].copy() 

181 good_bpms = data.iloc[np.where(labels != -1)].copy() 

182 bad_bpms_scores = iforest.decision_function(features.iloc[np.where(labels == -1)]) 

183 return bad_bpms, good_bpms, bad_bpms_scores

186def get_data_for_clustering( 

187 bpm_tfs_data: pd.DataFrame, plane: str, accelerator: Accelerator, 

188) -> tuple[pd.DataFrame, pd.DataFrame]: 

189 """Splits BPM data into arc and IR subsets and normalises features for clustering. 

190 

191 Args: 

192 bpm_tfs_data: concatenated BPM data from all input files. 

193 plane: marking the horizontal or vertical plane, **X** or **Y**. 

194 accelerator: accelerator instance providing the element types mask. 

195 

196 Returns: 

197 A tuple of (arc_bpm_data, ir_bpm_data) with normalised features. 

198 """ 

199 arc_bpm_mask = accelerator.get_element_types_mask(bpm_tfs_data.NAME, types=["arc_bpm"]) 

200 ir_bpm_data_for_clustering = bpm_tfs_data.iloc[~arc_bpm_mask].copy() 

201 arc_bpm_data_for_clustering = bpm_tfs_data.iloc[arc_bpm_mask].copy() 

202 for col in [f"TUNE{plane}", "NOISE_SCALED", f"AMP{plane}"]: 

203 ir_bpm_data_for_clustering.loc[:, col] = _normalize_parameter(ir_bpm_data_for_clustering.loc[:, col]) 

204 arc_bpm_data_for_clustering.loc[:, col] = _normalize_parameter(arc_bpm_data_for_clustering.loc[:, col]) 

205 return arc_bpm_data_for_clustering, ir_bpm_data_for_clustering

208def _normalize_parameter(column_data: pd.Series) -> pd.Series: 

209 """Rescales a Series to the [0, 1] range via min-max normalisation.""" 

210 return (column_data - column_data.min()) / (column_data.max() - column_data.min())

213def remove_bad_bpms( 

214 tfs_dfs: Sequence[tfs.TfsDataFrame], bad_bpm_names: list[str], plane: str, 

215) -> list[tfs.TfsDataFrame]: 

216 """Removes flagged BPMs from all input file DataFrames and recalculates tune statistics. 

217 

218 Args: 

219 tfs_dfs: list of measurement DataFrames. 

220 bad_bpm_names: names of BPMs to remove. 

221 plane: marking the horizontal or vertical plane, **X** or **Y**. 

222 

223 Returns: 

224 A list of the input TfsDataFrames with bad BPMs filtered out. 

225 """ 

226 cleaned: list[tfs.TfsDataFrame] = [] 

227 for df in tfs_dfs: 

228 filtered_df = df.loc[~df.index.isin(bad_bpm_names)] 

229 filtered_df.headers[f"Q{PLANE_TO_NUM[plane]}"] = np.mean(filtered_df[f"TUNE{plane}"]) 

230 filtered_df.headers[f"Q{PLANE_TO_NUM[plane]}RMS"] = np.std(filtered_df[f"TUNE{plane}"]) 

231 cleaned.append(filtered_df) 

232 return cleaned
