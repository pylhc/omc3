1""" 

2Harpy Parallel Helpers 

3---------------------- 

4 

5Private helper module for harpy's per-bunch parallel orchestration: 

6 

7- Resource estimation (peak resident memory per bunch, available RAM, usable cores), 

8- Dispatched workers count strategy. 

9 

10The functions are kept free of side effects, aside from querying ``psutil`` for host RAM 

11and cores. The actual dispatch / orchestration (and the BLAS thread-capping around it) lives 

12in :func:`omc3.harpy.handler.analyse_bunches_parallel`. 

13"""

31# ----- Some constants ----- # 

32 

33_BYTES_FLOAT64 = 8 

34_BYTES_COMPLEX128 = 16 

35 

36# Default fraction of available RAM that the worker pool is allowed to use. 

37# This is meant to leave a little bit of margin. Can be tweaked at function call. 

38ALLOWED_RAM_PORTION = 0.85 # 85% to leave room for other processes etc. 

39 

40# Empirically measured floor on the optics1 server (64-core, 536 BPMs, lin-only path): 

41# Python + NumPy + libraries + one bunch of data + the SVD working set. Overestimating 

42# this value is safe (leads to fewer workers for parallelisation and no risk of being OOM). 

43_BASELINE_RSS_BYTES = int(1.5e9) # 1.5GB peak RSS for a bunch in the lowest consuming scenario 

44 

45# Generous upper bound on the number of frequency lines selected by the narrow mask. 

46# Its transient is negligible next to the baseline. See comments inside of the 

47# estimate_peak_rss_bytes_per_bunch function for details and how this plays a role. 

48_NARROW_MASK_BINS = 1 << 14 # equivalent to 2**14

53def usable_cores() -> int: 

54 """ 

55 Tries to return the number of logical cores actually available to this process. 

56 

57 Prefers approaches that honour the CPU affinity mask (taskset, cgroup cpuset, 

58 HPC pinning, ...) so a pinned process does not oversubscribe. Tries, in order: 

59 

60 - ``os.process_cpu_count`` (Python 3.13+, honours affinity and ``PYTHON_CPU_COUNT``), 

61 - ``os.sched_getaffinity`` (Linux only, honours affinity), 

62 - ``psutil.Process().cpu_affinity`` (Windows/Linux only, honours affinity), 

63 - ``os.cpu_count`` (total logical CPUs, ignores affinity). 

64 

65 Always returns at least 1. 

66 """ 

67 LOGGER.debug("Determining available logical CPU cores, attempting to honour CPU affinity.") 

68 if hasattr(os, "process_cpu_count"): # Python 3.13+, honours affinity + PYTHON_CPU_COUNT 

69 return os.process_cpu_count() or 1 

70 

71 # fmt: off 

72 try: 

73 return len(os.sched_getaffinity(0)) # Linux only, honours affinity | # ty:ignore[unresolved-attribute] 

74 except AttributeError: # macOS / Windows: no sched_getaffinity 

75 pass 

76 # fmt: on 

77 

78 try: # macOS has no affinity concept -> AttributeError 

79 return len(psutil.Process().cpu_affinity()) or 1 # works on Windows 

80 except (AttributeError, NotImplementedError, OSError): 

81 pass 

82 

83 LOGGER.debug( 

84 "Defaulting to number of logical cores, with no affinity. " 

85 "This can lead to overloading if the machine is shared with many other processes." 

86 ) 

87 return os.cpu_count() or 1

90def available_ram_bytes() -> int | None: 

91 """ 

92 Returns currently available RAM in bytes (memory that can be

99 return None

105def estimate_peak_rss_bytes_per_bunch(harpy_input: DotDict, n_bpms: int) -> int: 

106 """ 

107 Analytic estimate of the peak resident set size memory of a single 

108 ``run_per_bunch`` call, in bytes. This is the number of bytes we estimate 

109 one worker will peak at for a given bunch. 

110 

111 The dominant transient for this is the coefficient array built when calling 

112 :func:`omc3.harpy.frequency.windowed_padded_rfft` (which allocates memory 

113 via `np.dot(u, s_vt_freq[:, mask])`, plus its `np.abs` copy). 

114 

115 Its width is the number of selected frequency lines: the full `2 ** turn_bits` 

116 when `full_spectra` are written (default from the BB GUI in the CCC) or cleaning 

117 is disabled (all-ones mask), and a negligible narrow band otherwise. 

118 

119 It scales with ``turn_bits`` and ``n_bpms``, not with the number of turns in the 

120 measurement, for two reasons. First, the rfft zero-pads its input to a fixed length 

121 set by `2 ** (turn_bits + 1)`, so the measurement's turn count is overshadowed by the 

122 padding. Second, the SVD reduces the data to ``sing_val`` rows before that FFT, so the 

123 turns never enter the dominant ``(n_bpms, n_mask)`` array at all. 

124 

125 Args: 

126 harpy_input (DotDict): Harpy analysis settings from which to query ``turn_bits``, 

127 ``to_write`` and ``clean``. 

128 n_bpms (int): number of BPMs (use the pre-clean count as a safe upper bound). 

129 

130 Returns: 

131 Estimated peak resident set size (RSS) memory in bytes. 

132 """ 

133 # Neglect the measurement array length (6600 for ACD, 40 000 for ADT) vs zero padding 

134 padded: int = 1 << harpy_input.turn_bits # equivalent to 2 ** turn_bits, frequency bins 

135 

136 # Determine if we have the extra memory usage from full-spectra / no cleaning 

137 full: bool = ("full_spectra" in harpy_input.to_write) or (not harpy_input.clean) 

138 

139 # Width, in frequency bins, of the coefficient array dominating the memory peak, 

140 # i.e. how many rfft bins survive get_freq_mask() into the `coefs` array. 

141 # - full: the whole half spectrum (2**turn_bits) is materialised, when full_spectra is required. 

142 # - otherwise: get_freq_mask() keeps only narrow ± tolerance bands around the tunes / resonances, 

143 # a few thousand bins. We have set _NARROW_MASK_BINS to safely over-estimate that. 

144 n_mask: int = padded if full else _NARROW_MASK_BINS 

145 

146 # At the FFT peak we have two arrays of shape (n_bpms, n_mask) simultaneously: 

147 # - coefs = np.dot(u, s_vt_freq[:, mask]), a complex128 array 

148 # - np.abs(coefs), its float64 copy 

149 transient: int = n_bpms * n_mask * (_BYTES_COMPLEX128 + _BYTES_FLOAT64) 

150 

151 # Return the estimated peak + minimum memory occupancy (data loading etc) 

152 return _BASELINE_RSS_BYTES + transient

155def decide_n_workers( 

156 harpy_input: DotDict, 

157 n_bunches: int, 

158 n_bpms: int, 

159 *, 

160 requested: int = 0, 

161 margin: float = ALLOWED_RAM_PORTION, 

162) -> int: 

163 """ 

164 Choose the number of worker processes for per-bunch analysis. Each bunch is sent 

165 to a worker. This assumes the number of BLAS threads is limited (for `numpy.svd` 

166 operations for instance where it maxes out the threads and doesn't do much with 

167 it). 

168 

169 Note 

170 ---- 

171 The `requested` argument mirrors the `n_jobs` for the harpy parameters. Passing 

172 `0` selects automatically (all usable cores, RAM permitting); `1` runs serially 

173 (which is the old behaviour) and providing an integer value `N` uses up to `N` 

174 workers, still bounded by the same safety limits. In other words `N` can only 

175 throttle the pool *below* the automatic choice, never oversubscribe it. Requesting 

176 `N` explicitly is also what lifts the fallback described below. 

177 

178 The result is `max(1, min(cores_cap, n_bunches, ram_cap))`, where `cores_cap` is 

179 `usable_cores()` when automatic or `min(N, usable_cores())` when `N` is requested, 

180 and `ram_cap` is `floor(margin * available_ram / peak_rss_per_bunch)` (or `1` when 

181 the available RAM could not be determined and no explicit `N` was requested, as we 

182 do not guess a pool size we cannot verify). 

183 

184 Args: 

185 harpy_input (DotDict): Harpy analysis settings, forwarded to 

186 :func:`estimate_peak_rss_bytes_per_bunch` for the per-bunch RSS estimate. 

187 n_bunches (int): Number of bunches to process. Caps the pool since each bunch 

188 goes to at most one worker. 

189 n_bpms (int): Number of BPMs (use the pre-clean count as a safe upper bound), 

190 used for the per-bunch RSS estimate. 

191 requested (int): the `n_jobs` option passed to ``harpy``. See the note above. 

192 margin (float): fraction of available RAM the worker pool may occupy, leaving 

193 the remainder as headroom against under-estimating the peak and for other 

194 running processes. 

195 

196 Returns: 

197 The number of worker processes to use, always at least 1. 

198 """ 

199 # Determine a few parameters. A requested N only ever caps *below* the usable cores: 

200 # workers are CPU-bound (BLAS capped to 1 thread), so more workers than cores just 

201 # oversubscribes with no gain. Hence N can be throttled down but never oversubscribe. 

202 cores_cap: int = min(requested, usable_cores()) if requested > 0 else usable_cores() 

203 peak_rss: int = estimate_peak_rss_bytes_per_bunch(harpy_input, n_bpms) 

204 available_ram: int | None = available_ram_bytes() 

205 

206 # Determine the available RAM to use, including the safety margin. 

207 # If unknown RAM -> we cannot size the pool. An OOM-killed worker takes the whole pool 

208 # down with it, so the safe failure mode is to stay serial when deciding automatically. 

209 # However, an explicitly provided 'n_jobs' is trusted, as the user knows their machine 

210 # better than a detection which just failed. 

211 if available_ram is None: 

212 available_ram_str = "unknown" 

213 if requested > 0: 

214 ram_cap: int = n_bunches # do not constrain, the user asked for it 

215 ram_cap_str = "n/a (explicit n_jobs)" 

216 LOGGER.warning( 

217 "Could not determine the available RAM on this machine; proceeding with " 

218 f"the explicitly requested n_jobs={requested}." 

219 ) 

220 else: 

221 ram_cap = 1 

222 ram_cap_str = "1 (unknown RAM)" 

223 LOGGER.warning( 

224 "Could not determine the available RAM on this machine, falling back to a " 

225 "single worker. Provide n_jobs=N explicitly to parallelise anyway." 

226 ) 

227 else: 

228 ram_cap = int(margin * available_ram / peak_rss) 

229 available_ram_str, ram_cap_str = f"{available_ram / 1e9:.0f}GB", str(ram_cap) 

230 

231 # In case the estimated peak RSS is higher than available RAM, we warn the user 

232 if peak_rss > available_ram: 

233 LOGGER.warning( 

234 f"The estimated peak RSS (memory usage) of a bunch is {peak_rss / 1e9:.1f}GB. " 

235 f"The available detected RAM is {available_ram_str}. " 

236 "This might lead to slowdowns or crashes!" 

237 ) 

238 

239 # Compute the number of workers / jobs to dispatch, log it and return 

240 n_jobs: int = max(1, min(cores_cap, n_bunches, ram_cap)) 

241 LOGGER.info( 

242 f"Parallel harpy: n_jobs={n_jobs} (cores_cap={cores_cap}, bunches={n_bunches}, " 

243 f"ram_cap={ram_cap_str}, peak/bunch~{peak_rss / 1e9:.1f}GB, available~{available_ram_str})" 

244 ) 

245 return n_jobs
